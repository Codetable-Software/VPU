cmake_minimum_required(VERSION 3.16)
project(vpu VERSION 1.0.0 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)
set(VPU_C src/vpu.c src/core.c src/registers.c src/flags.c src/context.c src/memory.c src/cache.c src/buffer.c src/instruction.c src/decoder.c src/executor.c src/alu.c src/vector.c src/task.c src/queue.c src/scheduler.c)
set(VPU_RUNTIME runtime/dispatch.c runtime/worker.c runtime/scheduler.c runtime/memory.c runtime/runtime.c)
add_library(vpu STATIC ${VPU_C} ${VPU_RUNTIME})
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  target_sources(vpu PRIVATE asm/x86_64/context.S asm/x86_64/memory.S asm/x86_64/math.S asm/x86_64/vector.S)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  target_sources(vpu PRIVATE asm/arm64/context.S asm/arm64/memory.S asm/arm64/math.S asm/arm64/vector.S)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "riscv64")
  target_sources(vpu PRIVATE asm/riscv64/context.S asm/riscv64/memory.S asm/riscv64/math.S asm/riscv64/vector.S)
endif()
add_library(vpu_compute STATIC cpp/engine.cpp cpp/compute.cpp cpp/vector.cpp cpp/runtime.cpp)
target_link_libraries(vpu_compute PUBLIC vpu)
add_executable(vpu-run tools/vpu-run.c)
add_executable(vpu-info tools/vpu-info.c)
add_executable(vpu-dump tools/vpu-dump.c)
add_executable(vpu-debug tools/vpu-debug.c)
add_executable(vpu-as assembler/assembler.c assembler/lexer.c assembler/parser.c assembler/encoder.c)
foreach(t vpu-run vpu-info vpu-dump vpu-debug)
  target_link_libraries(${t} PRIVATE vpu)
endforeach()
set(EXAMPLES basic compute vector task)
foreach(e IN LISTS EXAMPLES)
  add_executable(example-${e} examples/${e}.c)
  target_link_libraries(example-${e} PRIVATE vpu)
endforeach()
set(TESTS test_core test_memory test_alu test_vector test_decoder test_scheduler test_runtime)
foreach(t IN LISTS TESTS)
  add_executable(${t} tests/${t}.c)
  target_link_libraries(${t} PRIVATE vpu)
  add_test(NAME ${t} COMMAND ${t})
endforeach()
enable_testing()
install(DIRECTORY include/ DESTINATION include)
install(TARGETS vpu ARCHIVE DESTINATION lib)
